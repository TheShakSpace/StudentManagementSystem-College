//updating operators 
$set: chnage field value
$inc: increase or decrease number
$unset: remove a fiels
$push: add a value to an array
$pull: remove a value from an array

//increase rahul's age by 1 
$inc:

db.students.updateOne(
    { name: "Rahul" },
    { $inc: ( age: 1 )}
)

//remove a filed directly
$unset

db.students.updateOne(
    { name: "Rahul" },
    { $unset: { email: "" }}
)

//add a value to an array
$push

db.students.updateOne(
    { name: "Ojas" },
    { $push: (skills: "MongoDB" )}
)

//remocve a value from an array
$pull

db.students.updateOne(
    { name: "Ojas" },
    { $pull: { skills: "Node.js"}}
)

//GROUP DOCUMENTS BY FIELD
db.students.aggregate({
    {
        $group: {
            _id: "$course",
            totalStudents: {
                $sum: 1
            }
        }
    }
})


//output
{_id: "MERN", totalStudents: 2}
{_id: "Java", totalStudents: 3}
{_id: "Python", totalStudents: 1}


$avg, $sum, $max and $min are accumulator operators - they only appear inside $group, where they reduce every  document in a group down to a single value 

//This is the same $sum operator used to count documents with $sum: 1 - but here it adds up an actual field's values instead. 

$max: 
//oldest student in each course:
db.students.aggregate({
    {
        $group: {
            _id: "$course",
            maximumAge: {
                $max: "$age"
            }
        }
    }
})

//$max scans evry document in a group and keeps only the largest value of the field you give it - here, the oldest aggregate

$min:
//youngest age in each course:
db.students.aggregate({
    {
        $group: {
            _id: "$course",
            minimumAge: {
                $min: "$age"
            }
        }
    }
})

//$min works exactly like $max, but keeps the smallest value in each group - here, the youngest aggregate

//SORT RESULTS
$SORT
//Sort results by age, descending:

db.students.aggregate({
    {
        $sort: {
            age: -1
        }
    }
})


//1 - ascending
// -1 - descending

$limit: // cap the number of RESULTS
//return only the first 3, after sorting oldest first:

db.students.aggregate({
    {
        $sort: {
            age: -1
        }
    },
    {
        $limit: 3
    }
})
